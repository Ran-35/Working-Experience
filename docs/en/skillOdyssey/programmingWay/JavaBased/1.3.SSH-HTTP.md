##### (1) SSHJ Method

**Steps**


* (2) Code Implementation
```java
// Use try-with-resources to automatically close SSHClient
try (SSHClient ssh = new SSHClient()) {
    
    // Set connection timeout to 30 seconds
    ssh.setConnectTimeout(30000);
    
    // Set other operation timeouts to 30 seconds
    ssh.setTimeout(30000);
    
    // Add HostKey verifier, here using PromiscuousVerifier, meaning no SSH host key verification
    ssh.addHostKeyVerifier(new PromiscuousVerifier());
    
    // Connect to the specified IP address
    ssh.connect("192.168.1.1");
    
    // Authenticate using the specified username and password
    ssh.authPassword("username", "password");
    
    // Use try-with-resources to start a session and ensure it is automatically closed when done
    try (Session session = ssh.startSession()) {
        
        // Execute the shell command "ls -l" to list detailed file information of the current directory
        final Session.Command cmd = session.exec("ls -l");
        
        // Wait for the command to finish executing
        cmd.join();
    }

    // Upload a local file to a remote path
    ssh.newSCPFileTransfer().upload(new FileSystemFile(localFilePath), remoteFilePath);

    // Download a remote file to a local path
    ssh.newSCPFileTransfer().download(remoteFilePath, new FileSystemFile(localFileDownloadPath));
}


```

* (2) Used as a client in the code
```java
// 1. Create and start the client
try (SshClient client = SshClient.setUpDefaultClient()) {
    client.start();
    
    // 2. Establish a session
    try (ClientSession session = client.connect("username", "192.168.1.1", 22)
            .verify(10000).getSession()) {
        
        // 3. Authentication
        session.addPasswordIdentity("password");
        session.auth().verify(10000);

        // 4. Execute the command
        ByteArrayOutputStream responseStream = new ByteArrayOutputStream();
        try (ChannelExec channel = session.createExecChannel("ls -l")) {
            channel.setOut(responseStream);
            channel.open().verify(10000);
            
            // Wait for the command to finish executing
            channel.waitFor(java.util.EnumSet.of(org.apache.sshd.client.channel.ClientChannelEvent.CLOSED), 0);
            System.out.println("Command Output: " + responseStream.toString());
        }

        String localFilePath = "path/to/local/file.txt";
        String remoteFilePath = "/remote/path/file.txt";
        
        // 5. File upload
        File localFile = new File(localFilePath);
        try (ScpClient scpClient = session.createScpClient()) {
            scpClient.upload(localFile, remoteFilePath);
        }
        
        // 6. File download
        try (ScpClient scpClient = session.createScpClient()) {
            scpClient.download(remoteFilePath, new File(localDownloadPath));
        }
    }
}
```

#### 2. HTTP File Download

##### (1) Native Java 8 Method

* Download the file from the specified URL address and save it to the specified local path.
```java
public void download(){
    // The URL can also be a local directory.(String savePath = "C:/downloads/generated_file.zip")
    String fileUrl = "https://example.com/file.zip";
    String savePath = "C:/downloads/file.zip";

    try (InputStream in = new URL(fileUrl).openStream()) {
        Files.copy(in, Paths.get(savePath), StandardCopyOption.REPLACE_EXISTING);
        System.out.println("Download complete！");
    } catch (Exception e) {
        e.printStackTrace();
    } 
}
```
##### (2) Enhanced native Java 8 (with timeout control)
```java
public void download(){
    // The URL can also be a local directory.(String savePath = "C:/downloads/generated_file.zip")
    String fileUrl = "https://example.com/file.zip";
    String savePath = "C:/downloads/file.zip";

    URL url = new URL(fileUrl);
    HttpURLConnection httpConn = (HttpURLConnection) url.openConnection();
    
    // Set the timeout period (milliseconds).
    httpConn.setConnectTimeout(5000);
    httpConn.setReadTimeout(5000);

    int responseCode = httpConn.getResponseCode();
    if (responseCode == HttpURLConnection.HTTP_OK) {
        try (BufferedInputStream in = new BufferedInputStream(httpConn.getInputStream());
                FileOutputStream fileOutputStream = new FileOutputStream(savePath)) {
            
            byte[] dataBuffer = new byte[1024];
            int bytesRead;
            while ((bytesRead = in.read(dataBuffer, 0, 1024)) != -1) {
                fileOutputStream.write(dataBuffer, 0, bytesRead);
            }
        }
    }
    httpConn.disconnect();
}
```

##### (4) Spring Boot Recommended Approach (Resource Encapsulation)
```java
@GetMapping("/download")
public ResponseEntity<Resource> downloadFile() throws IOException {
    // Assuming this is the path to the file you generated.
    Path path = Paths.get("/path/to/your/generated_file.xlsx");
    Resource resource = new UrlResource(path.toUri());

    return ResponseEntity.ok()
            .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + resource.getFilename() + "\"")
            .contentType(MediaType.APPLICATION_OCTET_STREAM)
            .body(resource);
}
```
```java
@GetMapping("/download-string")
public ResponseEntity<byte[]> downloadString() {
    // 1. The file content comes from the someMethodToGenerateString() method
    String fileContent = someMethodToGenerateString();
    String fileName = "generated_data.txt";

    // 2. Convert the String to a byte array (ensure UTF-8 encoding to avoid issues with Chinese characters)
    byte[] contentBytes = fileContent.getBytes(StandardCharsets.UTF_8);

    // 3. Build the response headers
    HttpHeaders headers = new HttpHeaders();
    // Set the download file name
    headers.setContentDispositionFormData("attachment", fileName);
    // Set the content type to binary stream
    headers.setContentType(MediaType.APPLICATION_OCTET_STREAM);

    return new ResponseEntity<>(contentBytes, headers, HttpStatus.OK);
}

```
##### (5) Native Servlet Method (Java 8 Basic Edition)
```java
public void doGet(HttpServletResponse response) throws IOException {
    File file = new File("/path/to/generated_file.pdf");
    
    // Set the response headers
    response.setContentType("application/pdf");
    response.setHeader("Content-Disposition", "attachment; filename=" + file.getName());
    response.setContentLength((int) file.length());

    // Use Java 8 Files.copy for fast file copying
    try (InputStream in = new FileInputStream(file)) {
        Files.copy(in, response.getOutputStream());
    }
}

```

```Java
public void doGet(HttpServletRequest request, HttpServletResponse response) throws IOException {
    // The file content comes from the someMethodToGenerateString() method
    String fileContent = someMethodToGenerateString();
    
    // Set the response headers
    response.setContentType("text/plain;charset=UTF-8");
    response.setHeader("Content-Disposition", "attachment; filename=\"data.txt\"");
    
    // Write the String directly through the response's output stream
    try (PrintWriter writer = response.getWriter()) {
        writer.write(fileContent);
        writer.flush();
    }
}
```
